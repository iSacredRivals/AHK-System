-- SACRED AHK VERSION 11.0 | KEY SYSTEM & DELTA LAUNCHER
local REQUIRED_KEY = "franxx"
local KEY_FILE = "SacredAHK_SavedKey.txt"
local CONFIG_FILE = "SacredAHK_Config.json"

local Players = game:GetService("Players")
local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- Invalida la key guardada anterior si cambio la requerida
pcall(function()
    if isfile and readfile then
        if isfile(KEY_FILE) then
            local savedKey = readfile(KEY_FILE)
            if savedKey ~= REQUIRED_KEY then
                delfile(KEY_FILE)
                if isfile(CONFIG_FILE) then delfile(CONFIG_FILE) end
            end
        end
    end
end)

local function launchScript()
    for _, old in ipairs(playerGui:GetChildren()) do
        if old.Name:match("SacredKeyUI") then old:Destroy() end
    end
    -- Cargar el script principal completo desde tu enlace de GitHub
    loadstring(game:HttpGet("https://raw.githubusercontent.com/iSacredRivals/UI/refs/heads/main/Ui.txt"))()
end

-- Verificar si la key ya estaba guardada y es valida
local alreadyValid = false
pcall(function()
    if isfile and readfile and isfile(KEY_FILE) then
        if readfile(KEY_FILE) == REQUIRED_KEY then
            alreadyValid = true
        end
    end
end)

if alreadyValid then
    launchScript()
    return
end

-- Limpiar UIs antiguas de Key
for _, old in ipairs(playerGui:GetChildren()) do
    if old.Name:match("SacredKeyUI") then old:Destroy() end
end

-- Interfaz de Verificación de Key
local keyGui = Instance.new("ScreenGui")
keyGui.Name = "SacredKeyUI"
keyGui.ResetOnSpawn = false
keyGui.Parent = playerGui

local card = Instance.new("Frame", keyGui)
card.Size = UDim2.new(0, 320, 0, 200)
card.Position = UDim2.new(0.5, -160, 0.5, -100)
card.BackgroundColor3 = Color3.fromRGB(15, 12, 22)
card.BorderSizePixel = 0
card.Active = true
card.Draggable = true
Instance.new("UICorner", card).CornerRadius = UDim.new(0, 10)

local stroke = Instance.new("UIStroke", card)
stroke.Color = Color3.fromRGB(255, 255, 255)
stroke.Thickness = 1.5

local title = Instance.new("TextLabel", card)
title.Text = "🔑 SACRED AHK KEY SYSTEM"
title.Font = Enum.Font.GothamBold
title.TextSize = 13
title.TextColor3 = Color3.fromRGB(255, 255, 255)
title.Size = UDim2.new(1, 0, 0, 30)
title.Position = UDim2.new(0, 0, 0, 10)
title.BackgroundTransparency = 1

local input = Instance.new("TextBox", card)
input.Size = UDim2.new(0.85, 0, 0, 34)
input.Position = UDim2.new(0.075, 0, 0, 48)
input.BackgroundColor3 = Color3.fromRGB(25, 20, 35)
input.Text = ""
input.PlaceholderText = "Enter Key..."
input.Font = Enum.Font.GothamSemibold
input.TextSize = 11
input.TextColor3 = Color3.fromRGB(255, 255, 255)
input.PlaceholderColor3 = Color3.fromRGB(130, 130, 140)
Instance.new("UICorner", input).CornerRadius = UDim.new(0, 6)

local submit = Instance.new("TextButton", card)
submit.Size = UDim2.new(0.85, 0, 0, 32)
submit.Position = UDim2.new(0.075, 0, 0, 90)
submit.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
submit.Text = "VERIFY KEY"
submit.Font = Enum.Font.GothamBold
submit.TextSize = 11
submit.TextColor3 = Color3.fromRGB(15, 10, 20)
Instance.new("UICorner", submit).CornerRadius = UDim.new(0, 6)

local getDiscordBtn = Instance.new("TextButton", card)
getDiscordBtn.Size = UDim2.new(0.85, 0, 0, 28)
getDiscordBtn.Position = UDim2.new(0.075, 0, 0, 128)
getDiscordBtn.BackgroundColor3 = Color3.fromRGB(40, 30, 55)
getDiscordBtn.Text = "💬 Get Key (Discord)"
getDiscordBtn.Font = Enum.Font.GothamBold
getDiscordBtn.TextSize = 10
getDiscordBtn.TextColor3 = Color3.fromRGB(180, 160, 255)
Instance.new("UICorner", getDiscordBtn).CornerRadius = UDim.new(0, 6)

getDiscordBtn.MouseButton1Click:Connect(function()
    pcall(function() setclipboard("https://discord.gg/yBCXjavqw9") end)
    getDiscordBtn.Text = "✅ Discord Link Copied!"
    task.delay(1.5, function() getDiscordBtn.Text = "💬 Get Key (Discord)" end)
end)

local status = Instance.new("TextLabel", card)
status.Size = UDim2.new(1, 0, 0, 20)
status.Position = UDim2.new(0, 0, 0, 164)
status.BackgroundTransparency = 1
status.Text = ""
status.Font = Enum.Font.Gotham
status.TextSize = 9.5
status.TextColor3 = Color3.fromRGB(255, 80, 80)

submit.MouseButton1Click:Connect(function()
    if input.Text == REQUIRED_KEY then
        status.TextColor3 = Color3.fromRGB(0, 255, 120)
        status.Text = "✅ Key Verified! Loading Script..."
        pcall(function() if writefile then writefile(KEY_FILE, REQUIRED_KEY) end end)
        task.wait(0.5)
        launchScript()
    else
        status.TextColor3 = Color3.fromRGB(255, 80, 80)
        status.Text = "❌ Incorrect Key! Please check Discord."
    end
end)
