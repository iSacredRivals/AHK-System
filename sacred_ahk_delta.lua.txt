local REQUIRED_KEY = "bluebox"
local KEY_FILE = "SacredAHK_SavedKey.txt"
local CONFIG_FILE = "SacredAHK_Config.json"

local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")

local player = Players.LocalPlayer or Players:WaitForChildOfClass("Player")
local playerGui = player:WaitForChild("PlayerGui", 15)

if not playerGui then return end 

-- 1. Limpiar keys viejas de forma segura
pcall(function()
    if isfile and readfile and isfile(KEY_FILE) then
        local oldContent = readfile(KEY_FILE) or ""
        local cleanOld = string.lower(string.gsub(oldContent, "%s+", ""))
        if cleanOld ~= REQUIRED_KEY then
            if delfile then delfile(KEY_FILE) end
            if isfile(CONFIG_FILE) and delfile then delfile(CONFIG_FILE) end
        end
    end
end)

local SCRIPT_URL = "https://raw.githubusercontent.com/iSacredRivals/UI/main/Ui.txt"

local function launchScript()
    for _, old in ipairs(playerGui:GetChildren()) do
        if old.Name:match("SacredKeyUI") or old.Name:match("SacredUI") then old:Destroy() end
    end
    
    pcall(function()
        local rawScript = game:HttpGet(SCRIPT_URL)
        local fn, compileErr = loadstring(rawScript)
        if fn then
            fn()
        else
            warn("❌ Error compilando script de GitHub: " .. tostring(compileErr))
        end
    end)
end

-- 2. Verificar si ya está logueado de forma segura
local alreadyValid = false
pcall(function()
    if isfile and readfile and isfile(KEY_FILE) then
        local savedContent = readfile(KEY_FILE) or ""
        local cleanSaved = string.lower(string.gsub(savedContent, "%s+", ""))
        if cleanSaved == REQUIRED_KEY then 
            alreadyValid = true 
        end
    end
end)

if alreadyValid then
    launchScript()
    return
end

        status.TextColor3 = Color3.fromRGB(0, 255, 120)
        status.Text = isSpanish and "✅ ¡Key Correcta! Cargando..." or "✅ Key Verified! Loading..."
        
        -- Guardar de forma segura
        pcall(function() 
            if writefile then 
                writefile(KEY_FILE, REQUIRED_KEY) 
            end 
        end)
        
        task.wait(0.5)
        launcherGui:Destroy()
        launchScript()
    else
        -- Solo expulsar si realmente pusieron algo mal, no por error de archivo
        if userKey ~= "" then
            player:Kick(isSpanish and "❌ Key Incorrecta. Has sido expulsado." or "❌ Incorrect Key. You have been kicked.")
