local KEY_FILE = "SacredAHK_SavedKey.txt"
local Players = game:GetService("Players")
local player = Players.LocalPlayer
local playerGui = player:WaitForChild("PlayerGui")

-- Limpiar key vieja si existe
pcall(function()
    if isfile and readfile and isfile(KEY_FILE) then
        local savedKey = readfile(KEY_FILE) or ""
        local cleanSaved = string.lower(string.gsub(savedKey, "%s+", ""))
        local cleanRequired = string.lower(string.gsub("sacred", "%s+", ""))
        if cleanSaved ~= cleanRequired then
            if writefile then
                writefile(KEY_FILE, "")
            end
            if delfile then
                delfile(KEY_FILE)
            end
        end
    end
end)

local function launchScript()
    for _, old in ipairs(playerGui:GetChildren()) do
        if old.Name:match("SacredUI") then
            old:Destroy()
        end
    end
    local url = "https://raw.githubusercontent.com/iSacredRivals/AHK-/main/AHKMOBILE.lua"
    (loadstring(game:HttpGet(url)))()
end

local alreadyValid = false
pcall(function()
    if isfile and readfile and isfile(KEY_FILE) then
        local savedKey = readfile(KEY_FILE) or ""
        local cleanSaved = string.lower(string.gsub(savedKey, "%s+", ""))
        local cleanRequired = string.lower(string.gsub("sacred", "%s+", ""))
        if cleanSaved == cleanRequired and cleanSaved ~= "" then
            alreadyValid = true
        end
    end
end)

if alreadyValid then
    launchScript()
    return
end

for _, old in ipairs(playerGui:GetChildren()) do

    local requiredKey = string.lower(string.gsub("sacred", "%s+", ""))
    if userKey == requiredKey then
        statusLabel.TextColor3 = Color3.fromRGB(0, 255, 120)
        statusLabel.Text = "✅ Key Verified! Loading Script..."
        pcall(function()
            if writefile then
                writefile(KEY_FILE, "sacred")
            end
        end)
        task.wait(0.5)
        launchScript()
    else
        statusLabel.TextColor3 = Color3.fromRGB(255, 60, 60)
        statusLabel.Text = "❌ Incorrect Key! Check Discord."
    end
end)
